\section*{Answer Key with Explanations}

\subsection*{Question 1: Answer B}
$7 \times 8 = 56$. Basic multiplication.

\subsection*{Question 2: Answer C}
\begin{align*}
    5x &= 25 \\
    x &= \frac{25}{5} = 5
\end{align*}

\subsection*{Question 3: Answer C}
$\sqrt{144} = 12$ because $12 \times 12 = 144$

\subsection*{Question 4: Answer C}
\begin{align*}
    y &= 2x + 3 \\
    y &= 2(4) + 3 \\
    y &= 8 + 3 = 11
\end{align*}

\subsection*{Question 5: Answer C}
$$\frac{3}{4} + \frac{1}{4} = \frac{3+1}{4} = \frac{4}{4} = 1$$

\subsection*{Question 6: Answer C}
Perimeter of square = $4 \times \text{side} = 4 \times 5 = 20$ cm

\subsection*{Question 7: Answer D}
$2^5 = 2 \times 2 \times 2 \times 2 \times 2 = 32$

\subsection*{Question 8: Answer C}
In the equation $y = mx + b$, $m$ is the slope. Here $m = 3$.

\subsection*{Question 9: Answer C}
$$15\% \text{ of } 200 = \frac{15}{100} \times 200 = \frac{3000}{100} = 30$$

\subsection*{Question 10: Answer B}
Using the difference of squares formula:
$$(x + 2)(x - 2) = x^2 - 2^2 = x^2 - 4$$
